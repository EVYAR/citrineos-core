const assert=require('node:assert/strict'),fs=require('node:fs'),path=require('node:path'),{createHash}=require('node:crypto'),{spawnSync}=require('node:child_process');
const{readPrivateJson,readPrivateFile}=require('./private-path.cjs');
const root=path.resolve(__dirname,'../..'),sha=x=>createHash('sha256').update(x).digest('hex');
function sourcePin(){const r=spawnSync('git',['rev-parse','HEAD'],{cwd:root,encoding:'utf8'});assert.equal(r.status,0);return r.stdout.trim();}
function clean(){const r=spawnSync('git',['status','--porcelain'],{cwd:root,encoding:'utf8'});assert.equal(r.status,0);assert.equal(r.stdout.trim(),'','clean committed owner source required');}
function inventory(){const source=path.join(root,'apps/ocpp-server/migrations'),compiled=path.join(root,'apps/ocpp-server/dist/migrations'),names=fs.readdirSync(source).filter(n=>/^\d.*\.ts$/.test(n)).sort();assert.ok(names.includes('20261020210000-bind-status-ingress.ts'));assert.deepEqual(fs.readdirSync(compiled).filter(n=>/^\d.*\.js$/.test(n)).sort(),names.map(n=>n.replace(/\.ts$/,'.js')));return names.map(n=>({source:n,filename:n.replace(/\.ts$/,'.js'),sourceSha256:sha(fs.readFileSync(path.join(source,n))),compiledSha256:sha(fs.readFileSync(path.join(compiled,n.replace(/\.ts$/,'.js'))))}));}
function postgisEvidence(rows){assert.equal(rows.length,1,'exact PostGIS infrastructure extension required');const row=rows[0];assert.equal(row.extname,'postgis');assert.equal(row.schema,'public');assert.equal(typeof row.extversion,'string');assert.match(row.extversion,/^[0-9][0-9A-Za-z.+-]{0,63}$/);return{extname:row.extname,version:row.extversion,schema:row.schema};}
function assertEndpoint(value,host,name,user){let u;try{u=new URL(value);}catch{throw Error('PRIVATE_DATABASE_URL_INVALID');}assert.equal(u.protocol,'postgresql:');assert.equal(u.hostname,host);assert.equal(u.port||'5432','5432');assert.equal(u.pathname,'/'+name);assert.equal(decodeURIComponent(u.username),user);assert.ok(u.password);return u;}
function load(pathname){const r=readPrivateJson(pathname);clean();assert.equal(r.sourceCommit,sourcePin());assert.equal(r.stage,'CANONICAL_CORE_MIGRATIONS_READY');assert.equal(r.profile,'CORE_STATUS_INGRESS');assert.match(r.databaseName,/^cgp_core_status_[a-f0-9]{16}$/);assert.equal(r.role,r.databaseName+'_app');assert.equal(sha(readPrivateFile(r.inputPath)),r.inputSha256);const input=readPrivateJson(r.inputPath);assert.equal(input.databaseName,r.databaseName);assert.equal(input.sourceCommit,r.sourceCommit);assert.equal(input.receiptPath,pathname);assertEndpoint(input.maintenanceUrl,r.host,'postgres',r.adminUser);assert.equal(sha(readPrivateFile(input.buildProofPath)),r.buildProofSha256);const proof=readPrivateJson(input.buildProofPath);assert.equal(proof.head,r.sourceCommit);assert.equal(proof.exitCode,0);assert.deepEqual(inventory(),r.migrations);const app=assertEndpoint(r.serviceUrl,r.host,r.databaseName,r.role),admin=assertEndpoint(r.adminUrl,r.host,r.databaseName,r.adminUser);assert.notEqual(app.username,admin.username);return r;}
async function verify(client,r){const identity=(await client.query("SELECT current_database() AS database,current_user AS role,rolsuper,rolbypassrls,rolcreatedb,rolcreaterole,rolinherit,shobj_description(d.oid,'pg_database') AS marker,shobj_description(p.oid,'pg_authid') AS role_marker FROM pg_roles p JOIN pg_database d ON datname=current_database() WHERE rolname=current_user")).rows[0];assert.equal(identity.database,r.databaseName);assert.equal(identity.role,r.role);for(const k of ['rolsuper','rolbypassrls','rolcreatedb','rolcreaterole','rolinherit'])assert.equal(identity[k],false);assert.equal(identity.marker,r.marker+':'+r.ownershipNonce);assert.equal(identity.role_marker,r.marker+':'+r.ownershipNonce);assert.equal((await client.query('SELECT 1 FROM pg_auth_members WHERE member=(SELECT oid FROM pg_roles WHERE rolname=current_user)')).rowCount,0);const rows=(await client.query('SELECT name FROM "SequelizeMeta" ORDER BY name')).rows;assert.deepEqual(rows.map(x=>x.name),r.migrations.map(x=>x.filename));assert.deepEqual(postgisEvidence((await client.query("SELECT e.extname,e.extversion,n.nspname AS schema FROM pg_extension e JOIN pg_namespace n ON n.oid=e.extnamespace WHERE e.extname='postgis'")).rows),r.infrastructureExtensions.postgis);const acl=(await client.query("SELECT has_database_privilege(current_database(),'CREATE') AS create,has_database_privilege(current_database(),'TEMP') AS temp,has_schema_privilege('public','CREATE') AS schema_create")).rows[0];assert.deepEqual(acl,{create:false,temp:false,schema_create:false});}
module.exports={root,sha,sourcePin,clean,inventory,assertEndpoint,postgisEvidence,load,verify};
